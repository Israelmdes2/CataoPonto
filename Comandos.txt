# 1. Login no Sistema Senior por Imagem (OpenCV + PyAutoGUI)
.venv\Scripts\python.exe manage.py senior-login

.venv\Scripts\python.exe manage.py senior-login --user "seu_usuario" --password "sua_senha"

.venv\Scripts\python.exe manage.py senior-login --no-abrir

.venv\Scripts\python.exe manage.py senior-login --ambiente producao

# 2. Apenas abrir o Sistema Senior
.venv\Scripts\python.exe manage.py senior-open --ambiente homologacao
.venv\Scripts\python.exe manage.py senior-open --ambiente producao
